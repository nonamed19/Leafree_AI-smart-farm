import com.leafree.app.model.CropInfo
import com.leafree.app.model.EventResponse
import com.leafree.app.model.SettingInfo
import okhttp3.ResponseBody
import retrofit2.Response
import retrofit2.http.*

/**
 * Leafree 백엔드 서버와의 REST API 통신을 위한 인터페이스
 * 모든 API 엔드포인트는 https://leafree.kr/api를 기본 URL로 사용
 * 기기는 deviceId = 1 하나라 일단 mvp에서는 하드코딩
 */
interface LeafreeApi {
    companion object {
        const val BASE_URL = "https://leafree.kr/api/"
    }

    @POST("auth/login")
    suspend fun login(@Body credentials: LoginRequest): LoginResponse

    data class LoginRequest(
        val email: String,
        val password: String
    )

    data class LoginResponse(
        val userId: String,
        val accessToken: String
    )

    @GET("device/{deviceId}")
    suspend fun getCropInfo(@Path("deviceId") deviceId: Long): CropInfo

    @GET("device/{deviceId}/image")
    suspend fun getCropImage(@Path("deviceId") deviceId: Long): ResponseBody

    @GET("device/{deviceId}/setting")
    suspend fun getSettings(@Path("deviceId") deviceId: Long): SettingInfo

    @PATCH("device/{deviceId}/setting")
    suspend fun updateSetting(
        @Path("deviceId") deviceId: Long,
        @Body setting: SettingInfo
    )

    @GET("device/{deviceId}/event")
    suspend fun getEvents(@Path("deviceId") deviceId: Long, @Query("date") date: String): List<EventResponse>

    @POST("/api/device/{deviceId}/fcm-token")
    suspend fun sendFcmToken(
        @Path("deviceId") deviceId: Long,
        @Body token: FcmTokenRequest
    ): Response<Unit>

    data class FcmTokenRequest(
        val fcmToken: String
    )

}

package com.leafree.app.di

import android.content.Context
import com.leafree.app.api.LeafreeApi
import com.leafree.app.network.AuthInterceptor
import com.leafree.app.util.TokenManager
import okhttp3.OkHttpClient
import okhttp3.logging.HttpLoggingInterceptor
import retrofit2.Retrofit
import retrofit2.converter.gson.GsonConverterFactory

object NetworkModule {
    private var retrofit: Retrofit? = null
    private var api: LeafreeApi? = null

    fun provideApi(context: Context): LeafreeApi {
        if (api != null) return api!!

        val tokenManager = TokenManager(context)
        val logging = HttpLoggingInterceptor().apply { 
            level = HttpLoggingInterceptor.Level.BODY 
        }

        val client = OkHttpClient.Builder()
            .addInterceptor(AuthInterceptor(tokenManager))
            .addInterceptor(logging)
            .build()

        retrofit = Retrofit.Builder()
            .baseUrl(LeafreeApi.BASE_URL)
            .client(client)
            .addConverterFactory(GsonConverterFactory.create())
            .build()

        api = retrofit!!.create(LeafreeApi::class.java)
        return api!!
    }
}

package com.leafree.app.network

import com.leafree.app.util.TokenManager
import okhttp3.Interceptor
import okhttp3.Response

class AuthInterceptor(private val tokenManager: TokenManager) : Interceptor {
    override fun intercept(chain: Interceptor.Chain): Response {
        val token = tokenManager.getToken()
        val requestBuilder = chain.request().newBuilder()
        token?.let {
            requestBuilder.addHeader("Authorization", "Bearer $it")
        }
        return chain.proceed(requestBuilder.build())
    }
}

package com.leafree.app.network

import android.content.Context
import android.util.Log
import org.eclipse.paho.android.service.MqttAndroidClient
import org.eclipse.paho.android.service.MqttAndroidClient.Ack
import org.eclipse.paho.client.mqttv3.IMqttActionListener
import org.eclipse.paho.client.mqttv3.IMqttDeliveryToken
import org.eclipse.paho.client.mqttv3.IMqttToken
import org.eclipse.paho.client.mqttv3.MqttCallback
import org.eclipse.paho.client.mqttv3.MqttClient
import org.eclipse.paho.client.mqttv3.MqttConnectOptions
import org.eclipse.paho.client.mqttv3.MqttMessage

class MqttClientManager(
    private val context: Context,
    private val brokerUrl: String = "tcp://leafree.kr:1883",
    private val clientId: String = MqttClient.generateClientId(),
    private val username: String = "android_user",
    private val user_password: String = "leafree",
    private val topic: String = "/leafree/#"
    ) {

    private val mqttClient = MqttAndroidClient(context, brokerUrl, clientId, Ack.AUTO_ACK)

    fun connectAndSubscribe(onMessage: (message: String, topic: String) -> Unit) {
        val options = MqttConnectOptions().apply {
            isCleanSession = true
            setUserName(username)
            password = user_password.toCharArray()
            isAutomaticReconnect = true
            connectionTimeout = 30
            keepAliveInterval = 60
        }

        mqttClient.setCallback(object : MqttCallback {
            override fun messageArrived(topic: String?, message: MqttMessage?) {
                if (topic != null && message != null) {
                    onMessage(message.toString(), topic)
                }
            }

            override fun connectionLost(cause: Throwable?) {
                Log.e("MQTT", "연결 끊김: ${cause?.message}")
            }

            override fun deliveryComplete(token: IMqttDeliveryToken?) {}
        })

        mqttClient.connect(options, null, object : IMqttActionListener {
            override fun onSuccess(asyncActionToken: IMqttToken?) {
                Log.d("MQTT", "연결 성공")
                mqttClient.subscribe(topic, 1)
            }

            override fun onFailure(asyncActionToken: IMqttToken?, exception: Throwable?) {
                Log.e("MQTT", "연결 실패: ${exception?.message}")
            }
        })
    }


    fun disconnect() {
        mqttClient.disconnect()
    }

}

    override fun onMessageReceived(remoteMessage: RemoteMessage) {
        Log.d("FCM TEST", "알림 받기 성공")

        remoteMessage.notification?.let {
            val title = it.title
            val body = it.body
            val message = body.orEmpty()

            saveNotification(message) // 💾 알림 저장
            sendNotification(title, body) // 🔔 시스템 알림 표시
            broadcastNotification(message) // 🔁 앱 포그라운드면 UI 갱신
        }
    }

    private fun sendNotification(title: String?, body: String?) {
        Log.d("FCM TEST", "화면에 표시하기 직전")

        val intent = Intent(this, MainActivity::class.java).apply {
            addFlags(Intent.FLAG_ACTIVITY_CLEAR_TOP)
        }

        val pendingIntent = PendingIntent.getActivity(
            this, 0, intent,
            PendingIntent.FLAG_ONE_SHOT or PendingIntent.FLAG_IMMUTABLE
        )

        val channelId = "Notification"
        val defaultSoundUri: Uri = RingtoneManager.getDefaultUri(RingtoneManager.TYPE_NOTIFICATION)

        val notificationBuilder = NotificationCompat.Builder(this, channelId)
            .setSmallIcon(R.drawable.leafree_icon)
            .setContentTitle(title)
            .setContentText(body)
            .setAutoCancel(true)
            .setPriority(NotificationCompat.PRIORITY_HIGH)
            .setDefaults(NotificationCompat.DEFAULT_ALL)
            .setSound(defaultSoundUri)
            .setContentIntent(pendingIntent)

        val notificationManager =
            getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager

        val channel = NotificationChannel(
            channelId,
            "Leafree 알림 채널",
            NotificationManager.IMPORTANCE_HIGH
        ).apply {
            enableLights(true)
            enableVibration(true)
            setSound(
                RingtoneManager.getDefaultUri(RingtoneManager.TYPE_NOTIFICATION),
                null
            )
        }
        notificationManager.createNotificationChannel(channel)
        notificationManager.notify(0, notificationBuilder.build())
        Log.d("FCM TEST", "화면에 표시하기까지 성공")

    }

    private fun saveNotification(message: String) {
        val prefs = SharedPreferencesHelper(applicationContext)
        val currentList = prefs.getNotificationList()
        currentList.add(0, message) // 최신순
        prefs.saveNotificationList(currentList)

        val intent = Intent("com.leafree.NEW_NOTIFICATION").apply {
            putExtra("message", message)
            setPackage(packageName)
        }
        sendBroadcast(intent)
    }

import android.content.Context

import java.io.File
import java.util.UUID

class CropImageRepository {

    /**
     * 서버로부터 받은 byte[] 이미지를 내부 저장소에 저장하고 URI로 반환
     */
    fun saveImageToInternalStorage(context: Context, imageBytes: ByteArray): Uri {
        val uniqueFileName = "crop_${UUID.randomUUID()}.jpg"
        val file = File(context.filesDir, uniqueFileName)
        file.writeBytes(imageBytes)
        return file.toUri()
    }

    /**
     * 저장된 이미지 파일을 직접 삭제하거나 정리할 수 있도록 헬퍼 함수 제공 (옵션)
     */
    fun deleteImageFile(context: Context, fileName: String) {
        val file = File(context.filesDir, fileName)
        if (file.exists()) {
            file.delete()
        }
    }

    /**
     * 저장된 파일의 전체 목록을 확인하거나 디버깅용으로 사용 가능
     */
    fun listSavedImages(context: Context): List<File> {
        return context.filesDir.listFiles()?.toList() ?: emptyList()
    }
}

package com.leafree.app.util

import java.time.LocalDateTime
import java.time.ZoneId
import java.time.format.DateTimeFormatter
import java.util.Locale

/**
 * ISO 8601 UTC 형식 문자열을 LocalDateTime으로 변환
 * 예: "2025-05-19T07:01:38.846Z" → LocalDateTime
 */
fun parseUtcToLocalDateTime(utcString: String): LocalDateTime {
    val formatter = DateTimeFormatter.ofPattern("yyyy-MM-dd'T'HH:mm:ss")
    val localDateTime = LocalDateTime.parse(utcString, formatter)
    return localDateTime.atZone(ZoneId.of("UTC")).withZoneSameInstant(ZoneId.systemDefault()).toLocalDateTime()
}

/**
 * LocalDateTime에서 시간만 "오전/오후 h:mm" 형식으로 포맷
 */
fun formatTimeKorean(localDateTime: LocalDateTime): String {
    val formatter = DateTimeFormatter.ofPattern("a h:mm", Locale.KOREA)
    return localDateTime.toLocalTime().format(formatter)
}

fun formatDateKorean(localDateTime: LocalDateTime): String {
    val formatter = DateTimeFormatter.ofPattern("yyyy년 M월 d일", Locale.KOREA)
    return localDateTime.toLocalDate().format(formatter)

package com.leafree.app.util

import android.content.Context
import android.content.SharedPreferences
import com.google.gson.Gson
import com.google.gson.reflect.TypeToken

class SharedPreferencesHelper(context: Context) {
    private val prefs: SharedPreferences =
        context.getSharedPreferences("leafree_prefs", Context.MODE_PRIVATE)

    private val gson = Gson()

    // 🔐 [자동 로그인 토큰 저장/조회]
    fun saveToken(token: String) {
        prefs.edit().putString("access_token", token).apply()
    }

    fun getToken(): String? {
        return prefs.getString("access_token", null)
    }

    fun clearToken() {
        prefs.edit().remove("access_token").apply()
    }

    // 🔔 [알림 리스트 저장/조회]
    fun saveNotificationList(notifications: List<String>) {
        val json = gson.toJson(notifications)
        prefs.edit().putString("notification_list", json).apply()
    }

    fun getNotificationList(): MutableList<String> {
        val json = prefs.getString("notification_list", null)
        return if (json != null) {
            val type = object : TypeToken<MutableList<String>>() {}.type
            gson.fromJson(json, type)
        } else {
            mutableListOf()
        }
    }

    fun clearNotificationList() {
        prefs.edit().remove("notification_list").apply()
    }
}
