```C:\Users\Chanho Kim_PC\Desktop\project\Leafree_AI_Smart_Farm\LLM\app_LLM\endpoints.py
from fastapi import APIRouter, HTTPException
from fastapi.responses import FileResponse
from app_llm.plant_data_creator import PlantDataCreator
from app_llm.plant_image_creator import PlantImageCreator
from pathlib import Path
import json

router = APIRouter()
data_creator = PlantDataCreator()
img_creator = PlantImageCreator()

DATA_PATH = Path(__file__).parent / "datasets.json"
IMG_DIR = Path(__file__).parent / "images"
IMG_DIR.mkdir(exist_ok=True)

# GET: environmental condition data
@router.get("/data/{plant_name}")
async def get_plant_data(plant_name: str):
    plant_name = plant_name.strip()
    
    # Load dataset
    if DATA_PATH.exists():
        with open(DATA_PATH, "r", encoding="utf-8") as f:
            plant_datasets = json.load(f)
    else:
        plant_datasets = []

    # DATA SEARCH
    matched = next((i for i, p in enumerate(plant_datasets) if p["plant_name"].lower() == plant_name.lower()), None)
    
    if matched is not None:
        raw_data = plant_datasets[matched]
    else:
        raw_data = data_creator.plant_data(plant_name)

    # DATA PARSING
    updated_plant = {"plant_name": plant_name}
    updated_plant.update(raw_data)
    updated_plant["plant_name"] = plant_name

    if matched is not None:
        plant_datasets[matched] = updated_plant
    else:
        plant_datasets.append(updated_plant)

    # DATA VALIDATION CHECK
    if "error" in raw_data:
        raise HTTPException(status_code=404, detail="식물 이름이 잘못되었습니다.")

    with open(DATA_PATH, "w", encoding="utf-8") as f:
        json.dump(plant_datasets, f, ensure_ascii=False, indent=2)
    return {
        "plant_name": plant_name,
        "environment": raw_data,
    }

# POST: plant image
@router.post("/image/{plant_name}")
async def post_plant_image(plant_name: str):
    # # directory for images in docker
    DOCKER_DIR = IMG_DIR / f"{plant_name}.png"
    
    # # directory for images in EC2
    # EC2_DIR = Path("/app/images") / f"{plant_name}.png"

    if not DOCKER_DIR.exists():
        img_creator.create_image(plant_name, DOCKER_DIR)

    return FileResponse(
        path=DOCKER_DIR,
        media_type="image/png",
        filename=f"{plant_name}.png"
    )
```

```C:\Users\Chanho Kim_PC\Desktop\project\Leafree_AI_Smart_Farm\LLM\app_LLM\plant_image_creator.py
import os
import requests
from dotenv import load_dotenv
import openai

class PlantImageCreator:
    def __init__(self):
        load_dotenv()
        self.api_key = os.getenv("OPENAI_API_KEY")
        openai.api_key = self.api_key

    def create_image(self, plant_name, save_path):
        prompt = (
            "You are an expert caricature artist.\n"
            f"You will guess the plant name from the given sentence: {plant_name}.\n"
            "Create a cute, cartoon-style drawing of the plant.\n"
            "The plant should appear exactly once, positioned precisely at the center of the image.\n"
            "It should be designed as a simple, friendly character suitable with small facial features like eyes and a smile.\n"
            "The background must be a solid, light color with no patterns, textures, decorations, or additional elements—completely plain."
        )
        response = openai.images.generate(
            prompt = prompt,
            model = "dall-e-3",
            n = 1,
            size = "1024x1024",
            # quality = "hd",
        )

        image_url = response.data[0].url
        img_data = requests.get(image_url).content

        with open(save_path, 'wb') as handler:
            handler.write(img_data)
```

```C:\Users\Chanho Kim_PC\Desktop\project\Leafree_AI_Smart_Farm\LLM\OpenAI_API\GPT_based_bigdata.py
            {"role": "user", "content": f"Provide the optimal environmental data for '{plant}'."}

        functions=[plant_data_schema],
```

```C:\Users\Chanho Kim_PC\Desktop\project\Leafree_AI_Smart_Farm\LLM\plant_creators\Document_Parser\web_crawling\drfull_crawler.py
    plant_dataset[plant_name] = plant_info
    time.sleep(0.5)

# 3. crawling이 끝나면 파일을 json 형태로 변환 후 파일로 저장하기
```

```C:\Users\Chanho Kim_PC\Desktop\project\Leafree_AI_Smart_Farm\LLM\web_crawling\drfull_crawler.py
from bs4 import BeautifulSoup
import requests
import json
import time

# website URL
url = 'https://www.fuleaf.com/plants'
response = requests.get(url)
soup = BeautifulSoup(response.text, 'html.parser')

# search for <h3> tags
h3_tags = soup.find_all('h3')
plant_names = [tag.get_text(strip=True) for tag in h3_tags]
# 0. 첫 2개는 임의로 제거(전처리리)
plant_names = plant_names[2:] if len(plant_names) > 2 else plant_names

# 1. data셋 생성
plant_dataset = {}

# 2. 웹사이트 crawling을 통해 json data 형식으로 자료 저장
for idx, plant_name in enumerate(plant_names, 1):
    print(f"<{plant_name}> crawling ...")
    detail_url = f"https://drfull.im/plants/detail/{plant_name}"
    response = requests.get(detail_url)

    if response.status_code != 200:
        print(f"Status Code: {response.status_code})")
        continue

    plant_soup = BeautifulSoup(response.text, 'html.parser')

    plant_info = {
        'plant_english': None,
        'plant_intro': None,
        'water_period': None,
        'sunlight_period': None,
        'humidity': None,
        'temperature': None,
    }

    # search for simpleinfo-plantname_english mainColorTitle
    english_tag = plant_soup.find('p', class_='simpleinfo-plantname_english mainColorTitle')
    if english_tag:
        plant_info['plant_english'] = english_tag.get_text(strip=True)

    # search for simpleinfo-plantname_intro
    intro_tag = plant_soup.find('p', class_='simpleinfo-plantname_intro')
    if intro_tag:
        plant_info['plant_intro'] = intro_tag.get_text(strip=True)

    # search for details__tab-title
    tab_titles = plant_soup.find_all('p', class_='details__tab-title')
    if tab_titles:
        tab_title_temp = ['water_period', 'sunlight_period', 'humidity', 'temperature']
        for tab_title, key in zip(tab_titles, tab_title_temp):
            plant_info[key] = tab_title.get_text(strip=True)
    
    plant_dataset[plant_name] = plant_info
    time.sleep(0.5)

# 3. crawling이 끝나면 파일을 json 형태로 변환 후 파일로 저장하기
with open('plant_dataset.json', 'w', encoding='utf-8') as f:
    json.dump(plant_dataset, f, ensure_ascii=False, indent=4)

print("Crawling completed and data saved to plant_dataset.json")
```
