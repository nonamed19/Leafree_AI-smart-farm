```C:\Users\Chanho Kim_PC\Desktop\project\Leafree_AI_Smart_Farm\EM\Air_Solution\Core\Src\BH1750.c
//
//	Set the Measurement Time register. It allows to increase or decrease the sensitivity.
//
BH1750_STATUS BH1750_SetMtreg(uint8_t Mtreg)
{
	HAL_StatusTypeDef retCode;
	if (Mtreg < 31 || Mtreg > 254) {
		return BH1750_ERROR;
	}

	Bh1750_Mtreg = Mtreg;

	uint8_t tmp[2];

	tmp[0] = (0x40 | (Mtreg >> 5));
	tmp[1] = (0x60 | (Mtreg & 0x1F));

	retCode = HAL_I2C_Master_Transmit(bh1750_i2c, BH1750_ADDRESS, &tmp[0], 1, 10);
	if (HAL_OK != retCode) {
		return BH1750_ERROR;
	}

	retCode = HAL_I2C_Master_Transmit(bh1750_i2c, BH1750_ADDRESS, &tmp[1], 1, 10);

		return BH1750_OK;
```

```C:\Users\Chanho Kim_PC\Desktop\project\Leafree_AI_Smart_Farm\EM\Air_Solution\Core\Src\DHT.c
#include "DHT.h"

#define lineDown() 		HAL_GPIO_WritePin(sensor->DHT_Port, sensor->DHT_Pin, GPIO_PIN_RESET)
#define lineUp()		HAL_GPIO_WritePin(sensor->DHT_Port, sensor->DHT_Pin, GPIO_PIN_SET)

static void goToOutput(DHT_sensor *sensor) {
	GPIO_InitTypeDef GPIO_InitStruct = {0};

  lineUp();

  GPIO_InitStruct.Pin = sensor->DHT_Pin;
  GPIO_InitStruct.Mode = GPIO_MODE_OUTPUT_OD;
  GPIO_InitStruct.Pull = sensor->pullUp;

  GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_HIGH;
  HAL_GPIO_Init(sensor->DHT_Port, &GPIO_InitStruct);
}

static void goToInput(DHT_sensor *sensor) {
  GPIO_InitTypeDef GPIO_InitStruct = {0};

  GPIO_InitStruct.Pin = sensor->DHT_Pin;
  GPIO_InitStruct.Mode = GPIO_MODE_INPUT;
  GPIO_InitStruct.Pull = sensor->pullUp;
  HAL_GPIO_Init(sensor->DHT_Port, &GPIO_InitStruct);
}

DHT_data DHT_getData(DHT_sensor *sensor) {
	DHT_data data = {-128.0f, -128.0f};

	#if DHT_POLLING_CONTROL == 1
	uint16_t pollingInterval;
	if (sensor->type == DHT11) {
		pollingInterval = DHT_POLLING_INTERVAL_DHT11;
	} else {
		pollingInterval = DHT_POLLING_INTERVAL_DHT22;
	}

	if ((HAL_GetTick() - sensor->lastPollingTime < pollingInterval) && sensor->lastPollingTime != 0) {
		data.hum = sensor->lastHum;
		data.temp = sensor->lastTemp;
		return data;
	}
	sensor->lastPollingTime = HAL_GetTick()+1;
	#endif

	goToOutput(sensor);
	lineDown();
	Delay(2);
	lineUp();
	goToInput(sensor);


	#ifdef DHT_IRQ_CONTROL
	__disable_irq();
	#endif
	uint16_t timeout = 0;
	while(getLine()) {
		timeout++;
		if (timeout > DHT_TIMEOUT) {
			#ifdef DHT_IRQ_CONTROL
			__enable_irq();
			#endif

			sensor->lastHum = -128.0f;
			sensor->lastTemp = -128.0f;

			return data;
		}
	}
	timeout = 0;
	while(!getLine()) {
		timeout++;
		if (timeout > DHT_TIMEOUT) {
			#ifdef DHT_IRQ_CONTROL
			__enable_irq();
			#endif

			sensor->lastHum = -128.0f;
			sensor->lastTemp = -128.0f;

			return data;
		}
	}
	timeout = 0;
	while(getLine()) {
		timeout++;
		if (timeout > DHT_TIMEOUT) {
			#ifdef DHT_IRQ_CONTROL
			__enable_irq();
			#endif
			return data;
		}
	}

	uint8_t rawData[5] = {0,0,0,0,0};
	for(uint8_t a = 0; a < 5; a++) {
		for(uint8_t b = 7; b != 255; b--) {
			uint16_t hT = 0, lT = 0;
			while(!getLine() && lT != 65535) lT++;
			timeout = 0;
			while(getLine()&& hT != 65535) hT++;
			if(hT > lT) rawData[a] |= (1<<b);
		}
	}

    #ifdef DHT_IRQ_CONTROL
	__enable_irq();
    #endif

	if((uint8_t)(rawData[0] + rawData[1] + rawData[2] + rawData[3]) == rawData[4]) {
		if (sensor->type == DHT22) {
			data.hum = (float)(((uint16_t)rawData[0]<<8) | rawData[1])*0.1f;
			if(!(rawData[2] & (1<<7))) {
				data.temp = (float)(((uint16_t)rawData[2]<<8) | rawData[3])*0.1f;
			}	else {
				rawData[2] &= ~(1<<7);
				data.temp = (float)(((uint16_t)rawData[2]<<8) | rawData[3])*-0.1f;
			}
		}
		if (sensor->type == DHT11) {
			data.hum = (float)rawData[0];
			data.temp = (float)rawData[2];
		}
	}

	#if DHT_POLLING_CONTROL == 1
	sensor->lastHum = data.hum;
	sensor->lastTemp = data.temp;
	#endif

	return data;
}
```

```C:\Users\Chanho Kim_PC\Desktop\project\Leafree_AI_Smart_Farm\EM\Air_Solution\Core\Src\freertos.c
void StartLightControlTask(void *argument)
{
    WS2812B_Init();
    int start_hour = 8;
    int end_hour = start_hour + std_light_hours;

    for (;;)
    {
        uint8_t hour = sTime.Hours;
        bool in_light_period = false;

        // 조광 시간이 1 이상인 경우만 판단
        if (std_light_hours > 0)
        {
            if (hour >= start_hour && hour < end_hour)
            {
                in_light_period = true;
            }
        }

        if (in_light_period && g_lux < std_lux)
        {
            // 조도가 부족할 때만 생장등 ON
            for (int i = 0; i < WS2812_LED_COUNT; i++)
                WS2812B_SetColor(i, 255, 0, 50);  // 핑크색
        }
        else
        {
            // 그 외에는 OFF
            for (int i = 0; i < WS2812_LED_COUNT; i++)
                WS2812B_SetColor(i, 0, 0, 0);     // OFF
        }
        WS2812B_Show();
        osDelay(1000);
    }
}

void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart)
{
    if (huart == &huart3)
    {
        BaseType_t hp = pdFALSE;
        uint8_t cmd = rxBuf[0];

        if (cmd == 0x01 || cmd == 0x04)  // 센서 제어
        {
            AirCtl_t pkt;
            memcpy(&pkt, rxBuf, sizeof(AirCtl_t));
            xQueueSendFromISR(xAirCtlQ, &pkt, &hp);
        }
        else if (cmd == 0x02)  // RTC 설정
        {
            RtcSetPacket_t rtc;
            memcpy(&rtc, rxBuf, sizeof(RtcSetPacket_t));
            xQueueSendFromISR(xRtcSetQ, &rtc, &hp);
        }

        portYIELD_FROM_ISR(hp);

        // 다음 수신 준비 (어떤 패킷이 올지 모르니 가장 큰 것으로)
        HAL_UART_Receive_DMA(&huart3, rxBuf, sizeof(rxBuf));
    }
}

    AirCtl_t airPkt;
    RtcSetPacket_t rtcPkt;

    for (;;)
    {
        if (xQueueReceive(xAirCtlQ, &airPkt, 0) == pdPASS)

            if (airPkt.cmd == 0x01)
            {
                std_temp     = (float)airPkt.temp10 / 10.0f;
                std_humidity = airPkt.humidity;
                std_lux      = airPkt.lux;
                std_co2 = airPkt.co2;

            }
            else if (airPkt.cmd == 0x04)
            {
                AirStat_t stat;
                stat.temp10  = (int16_t)(g_temp * 10.0f);
                stat.humidity = (uint8_t)g_humidity;
                stat.lux      = (uint16_t)g_lux;
                stat.co2 = (uint16_t)g_co2;

                HAL_UART_Transmit(&huart3, (uint8_t*)&stat, sizeof(stat), HAL_MAX_DELAY);
//

//                         g_temp, g_humidity, g_lux);
//                HAL_UART_Transmit(&huart2, (uint8_t*)msg, strlen(msg), HAL_MAX_DELAY);
            }
        }
        if (xQueueReceive(xRtcSetQ, &rtcPkt, 0) == pdPASS)
        {
            RTC_TimeTypeDef sTime;
            RTC_DateTypeDef sDate;

            sTime.Hours = rtcPkt.hour;
            sTime.Minutes = rtcPkt.minute;
            sTime.Seconds = rtcPkt.second;

            sDate.Date = rtcPkt.day;
            sDate.Month = rtcPkt.month;
            sDate.Year = rtcPkt.year;

            HAL_RTC_SetTime(&hrtc, &sTime, RTC_FORMAT_BIN);
            HAL_RTC_SetDate(&hrtc, &sDate, RTC_FORMAT_BIN);

        }

        osDelay(10);  // 짧게 sleep 주기
    }
```

```C:\Users\Chanho Kim_PC\Desktop\project\Leafree_AI_Smart_Farm\EM\Air_Solution\Core\Src\stm32f1xx_hal_timebase_tim.c
/**
  * @brief  This function configures the TIM1 as a time base source.
  *         The time source is configured  to have 1ms time base with a dedicated
  *         Tick interrupt priority.
  * @note   This function is called  automatically at the beginning of program after
  *         reset by HAL_Init() or at any time when clock is configured, by HAL_RCC_ClockConfig().
  * @param  TickPriority: Tick interrupt priority.
  * @retval HAL status
  */
HAL_StatusTypeDef HAL_InitTick(uint32_t TickPriority)
{
  RCC_ClkInitTypeDef    clkconfig;
  uint32_t              uwTimclock = 0U;

  uint32_t              uwPrescalerValue = 0U;
  uint32_t              pFLatency;

  HAL_StatusTypeDef     status = HAL_OK;

  /* Enable TIM1 clock */
  __HAL_RCC_TIM1_CLK_ENABLE();

  /* Get clock configuration */
  HAL_RCC_GetClockConfig(&clkconfig, &pFLatency);

  /* Compute TIM1 clock */
      uwTimclock = HAL_RCC_GetPCLK2Freq();

  /* Compute the prescaler value to have TIM1 counter clock equal to 1MHz */
  uwPrescalerValue = (uint32_t) ((uwTimclock / 1000000U) - 1U);

  /* Initialize TIM1 */
  htim1.Instance = TIM1;

  /* Initialize TIMx peripheral as follow:
   * Period = [(TIM1CLK/1000) - 1]. to have a (1/1000) s time base.
   * Prescaler = (uwTimclock/1000000 - 1) to have a 1MHz counter clock.
   * ClockDivision = 0
   * Counter direction = Up
   */
  htim1.Init.Period = (1000000U / 1000U) - 1U;
  htim1.Init.Prescaler = uwPrescalerValue;
  htim1.Init.ClockDivision = 0;
  htim1.Init.CounterMode = TIM_COUNTERMODE_UP;
  htim1.Init.AutoReloadPreload = TIM_AUTORELOAD_PRELOAD_DISABLE;

  status = HAL_TIM_Base_Init(&htim1);
  if (status == HAL_OK)
  {
    /* Start the TIM time Base generation in interrupt mode */
    status = HAL_TIM_Base_Start_IT(&htim1);
    if (status == HAL_OK)
    {
    /* Enable the TIM1 global Interrupt */
        HAL_NVIC_EnableIRQ(TIM1_UP_IRQn);
      /* Configure the SysTick IRQ priority */
      if (TickPriority < (1UL << __NVIC_PRIO_BITS))
      {
        /* Configure the TIM IRQ priority */
        HAL_NVIC_SetPriority(TIM1_UP_IRQn, TickPriority, 0U);
        uwTickPrio = TickPriority;
      }
      else
      {
        status = HAL_ERROR;
      }
    }
  }

 /* Return function status */
  return status;
}

/**
```
```C:\Users\Chanho Kim_PC\Desktop\project\Leafree_AI_Smart_Farm\EM\Air_Solution\Core\Src\sysmem.c
 * @brief _sbrk() allocates memory to the newlib heap and is used by malloc
 *        and others from the C library
 *
 * @verbatim
 * ############################################################################
 * #  .data  #  .bss  #       newlib heap       #          MSP stack          #
 * #         #        #                         # Reserved by _Min_Stack_Size #
 * ############################################################################
 * ^-- RAM start      ^-- _end                             _estack, RAM end --^
 * @endverbatim
 *
 * This implementation starts allocating at the '_end' linker symbol
 * The '_Min_Stack_Size' linker symbol reserves a memory for the MSP stack
 * The implementation considers '_estack' linker symbol to be RAM end
 * NOTE: If the MSP stack, at any point during execution, grows larger than the
 * reserved size, please increase the '_Min_Stack_Size'.
 *
 * @param incr Memory size
 * @return Pointer to allocated memory
 */
void *_sbrk(ptrdiff_t incr)
{
  extern uint8_t _end; /* Symbol defined in the linker script */
  extern uint8_t _estack; /* Symbol defined in the linker script */
  extern uint32_t _Min_Stack_Size; /* Symbol defined in the linker script */
  const uint32_t stack_limit = (uint32_t)&_estack - (uint32_t)&_Min_Stack_Size;
  const uint8_t *max_heap = (uint8_t *)stack_limit;
  uint8_t *prev_heap_end;

  /* Initialize heap end at first call */
  if (NULL == __sbrk_heap_end)
  {
    __sbrk_heap_end = &_end;
  }

  /* Protect heap from growing into the reserved MSP stack */
  if (__sbrk_heap_end + incr > max_heap)
  {
    errno = ENOMEM;
    return (void *)-1;
  }

  prev_heap_end = __sbrk_heap_end;
  __sbrk_heap_end += incr;

  return (void *)prev_heap_end;
}
```

```C:\Users\Chanho Kim_PC\Desktop\project\Leafree_AI_Smart_Farm\EM\Air_Solution\Core\Src\system_stm32f1xx.c
  *         The SystemCoreClock variable contains the core clock (HCLK), it can
  *         be used by the user application to setup the SysTick timer or configure
  *         other parameters.
  *           
  * @note   Each time the core clock (HCLK) changes, this function must be called
  *         to update SystemCoreClock variable value. Otherwise, any configuration
  *         based on this variable will be incorrect.         
  *     
  * @note   - The system frequency computed by this function is not the real 
  *           frequency in the chip. It is calculated based on the predefined 
  *           constant and the selected clock source:
  *             
  *           - If SYSCLK source is HSI, SystemCoreClock will contain the HSI_VALUE(*)
  *                                              
  *           - If SYSCLK source is HSE, SystemCoreClock will contain the HSE_VALUE(**)
  *                          
  *           - If SYSCLK source is PLL, SystemCoreClock will contain the HSE_VALUE(**) 
  *             or HSI_VALUE(*) multiplied by the PLL factors.
  *         
  *         (*) HSI_VALUE is a constant defined in stm32f1xx.h file (default value
  *             8 MHz) but the real value may vary depending on the variations
  *             in voltage and temperature.   
  *    
  *         (**) HSE_VALUE is a constant defined in stm32f1xx.h file (default value
  *              8 MHz or 25 MHz, depending on the product used), user has to ensure
  *              that HSE_VALUE is same as the real frequency of the crystal used.
  *              Otherwise, this function may have wrong result.
  *                
  *         - The result of this function could be not correct when using fractional
  *           value for HSE crystal.
  * @param  None
  * @retval None
  */
void SystemCoreClockUpdate (void)
{
  uint32_t tmp = 0U, pllmull = 0U, pllsource = 0U;

#if defined(STM32F105xC) || defined(STM32F107xC)
  uint32_t prediv1source = 0U, prediv1factor = 0U, prediv2factor = 0U, pll2mull = 0U;
#endif /* STM32F105xC */

#if defined(STM32F100xB) || defined(STM32F100xE)
  uint32_t prediv1factor = 0U;
#endif /* STM32F100xB or STM32F100xE */
    
  /* Get SYSCLK source -------------------------------------------------------*/
  tmp = RCC->CFGR & RCC_CFGR_SWS;
  
  switch (tmp)
  {
    case 0x00U:  /* HSI used as system clock */
      SystemCoreClock = HSI_VALUE;
      break;
    case 0x04U:  /* HSE used as system clock */
      SystemCoreClock = HSE_VALUE;
      break;
    case 0x08U:  /* PLL used as system clock */

      /* Get PLL clock source and multiplication factor ----------------------*/
      pllmull = RCC->CFGR & RCC_CFGR_PLLMULL;
      pllsource = RCC->CFGR & RCC_CFGR_PLLSRC;
      
#if !defined(STM32F105xC) && !defined(STM32F107xC)      
      pllmull = ( pllmull >> 18U) + 2U;
      
      if (pllsource == 0x00U)
      {
        /* HSI oscillator clock divided by 2 selected as PLL clock entry */
        SystemCoreClock = (HSI_VALUE >> 1U) * pllmull;
      }
      else
      {
 #if defined(STM32F100xB) || defined(STM32F100xE)
       prediv1factor = (RCC->CFGR2 & RCC_CFGR2_PREDIV1) + 1U;
       /* HSE oscillator clock selected as PREDIV1 clock entry */
       SystemCoreClock = (HSE_VALUE / prediv1factor) * pllmull; 
 #else
        /* HSE selected as PLL clock entry */
        if ((RCC->CFGR & RCC_CFGR_PLLXTPRE) != (uint32_t)RESET)
        {/* HSE oscillator clock divided by 2 */
          SystemCoreClock = (HSE_VALUE >> 1U) * pllmull;
        }
        else
        {
          SystemCoreClock = HSE_VALUE * pllmull;
        }
 #endif
      }
#else
      pllmull = pllmull >> 18U;
      
      if (pllmull != 0x0DU)
      {
         pllmull += 2U;
      }
      else
      { /* PLL multiplication factor = PLL input clock * 6.5 */
        pllmull = 13U / 2U; 
      }
            
      if (pllsource == 0x00U)
      {
        /* HSI oscillator clock divided by 2 selected as PLL clock entry */
        SystemCoreClock = (HSI_VALUE >> 1U) * pllmull;
      }
      else
      {/* PREDIV1 selected as PLL clock entry */
        
        /* Get PREDIV1 clock source and division factor */
        prediv1source = RCC->CFGR2 & RCC_CFGR2_PREDIV1SRC;
        prediv1factor = (RCC->CFGR2 & RCC_CFGR2_PREDIV1) + 1U;
        
        if (prediv1source == 0U)
        { 
          /* HSE oscillator clock selected as PREDIV1 clock entry */
          SystemCoreClock = (HSE_VALUE / prediv1factor) * pllmull;          
        }
        else
        {/* PLL2 clock selected as PREDIV1 clock entry */
          
          /* Get PREDIV2 division factor and PLL2 multiplication factor */
          prediv2factor = ((RCC->CFGR2 & RCC_CFGR2_PREDIV2) >> 4U) + 1U;
          pll2mull = ((RCC->CFGR2 & RCC_CFGR2_PLL2MUL) >> 8U) + 2U; 
          SystemCoreClock = (((HSE_VALUE / prediv2factor) * pll2mull) / prediv1factor) * pllmull;                         
        }
      }
#endif /* STM32F105xC */ 
      break;

    default:
      SystemCoreClock = HSI_VALUE;
      break;
  }

  /* Get HCLK prescaler */
  tmp = AHBPrescTable[((RCC->CFGR & RCC_CFGR_HPRE) >> 4U)];
  /* HCLK clock frequency */
  SystemCoreClock >>= tmp;  
}
```

```C:\Users\Chanho Kim_PC\Desktop\project\Leafree_AI_Smart_Farm\EM\Air_Solution\Core\Src\WS2812B.c
void WS2812B_SetColor(uint32_t index, uint8_t r, uint8_t g, uint8_t b) {
    if (index >= WS2812_LED_COUNT) return;
    uint32_t offset = WS2812_RESET_SLOTS + index * 24;

        ws2812b_dma_buffer[offset + i] = (g & (1 << (7 - i))) ? 56 : 28;
        ws2812b_dma_buffer[offset + 8 + i] = (r & (1 << (7 - i))) ? 56 : 28;
        ws2812b_dma_buffer[offset + 16 + i] = (b & (1 << (7 - i))) ? 56 : 28;
    }
}
```

```C:\Users\Chanho Kim_PC\Desktop\project\Leafree_AI_Smart_Farm\EM\Interface\leafree\widget.cpp
void Widget::handle_control_message(const QMqttMessage& msg)
{
    const QString topic = msg.topic().name();
    const QString payloadStr = QString::fromUtf8(msg.payload());

    if (topic.endsWith("/temperature")) {
        ui->target_temp_value->setText(payloadStr);
        aflag |= 1;
    } else if (topic.endsWith("/humidity")) {
        ui->target_humid_value->setText(payloadStr);
        aflag |= (1 << 1);
    } else if (topic.endsWith("/light")) {
        ui->target_light_value->setText(payloadStr);
        aflag |= (1 << 2);
    } else if (topic.endsWith("/co2_ppm")) {
        ui->target_co2_value->setText(payloadStr);
        aflag |= (1 << 3);
    } else if (topic.endsWith("/light_duration")) {
        double value = payloadStr.toFloat();
        double result = value / 60.0;
        QString formatted = QString::number(result, 'f', 1);
        ui->target_duration_value->setText(formatted);
        aflag |= (1 << 4);

    } else if (topic.endsWith("/watering_interval")) {
        double value = payloadStr.toFloat();
        double result = value / 60.0;
        QString formatted = QString::number(result, 'f', 1);
        ui->target_interval_value->setText(formatted);
        wflag |= 1;
    } else if (topic.endsWith("/soil_moisture")) {
        ui->target_moist_value->setText(payloadStr);
        wflag |= (1 << 1);

    } else if (topic.endsWith("/crop_image")) {
        update_crop_image(msg); // 예외적으로 이미지 처리는 그대로 별도 함수 사용
    }

    if(aflag == 0b11111) {
        serial_sensor_transmit();
        aflag = 0;
    }

    if(wflag == 0b11) {
        serial_pump_transmit();
        wflag = 0;
    }
}

void Widget::serial_pump_received()
{
    pump_buffer.append(serial_pump->readAll());

    while(pump_buffer.size() >= 8) {
        QByteArray packet = pump_buffer.left(8);
        pump_buffer.remove(0, 8);

        QString hexString = packet.toHex(' ').toUpper();  // 16진수로 변환, 공백 구분
        ui->rx_pump_lbl->setText("received: " + hexString);

        SensorData_t value;
        value.soil_adc = packet[0] | (packet[1] << 8);
        value.level_adc = packet[2] | (packet[3] << 8);
        if(value.level_adc < WATER_LV_MIN) value.level_adc = WATER_LV_MIN;
        else if(value.level_adc > WATER_LV_MAX) value.level_adc = WATER_LV_MAX;
        value.sample_tick = packet[4] | (packet[5] << 8) | (packet[6] << 16) | (packet[7] << 24);

        ui->waterlv_value->setText(QString::number(((value.level_adc - WATER_LV_MIN) * 100.0) / (WATER_LV_MAX - WATER_LV_MIN), 'f', 1));
        ui->soilmoist_value->setText(QString::number(((SOIL_MOIST_MAX - value.soil_adc) * 100.0) / (SOIL_MOIST_MAX - SOIL_MOIST_MIN), 'f', 1));

        m_client->publish(MQTT_COMMON_TOPIC + "/sensor/water_level", ui->waterlv_value->text().toUtf8());
        m_client->publish(MQTT_COMMON_TOPIC + "/sensor/soil_moisture", ui->soilmoist_value->text().toUtf8());
    }

    //pump_buffer.clear();
}

void Widget::pipe_connect() {
    qDebug() << "pipeConnect called";
    QFile cmdPipe("/tmp/stt_cmd");
    if (cmdPipe.open(QIODevice::WriteOnly)) {
        QTextStream out(&cmdPipe);
        out << "START\n";
        cmdPipe.flush();
        cmdPipe.close();
    }

    int fd = open("/tmp/stt_resp", O_RDONLY | O_NONBLOCK);
    if (fd == -1) {
        qDebug() << "Failed to open /tmp/stt_resp non-blocking";
        return;
    }

    respPipe = new QFile(this);
    respPipe->open(fd, QIODevice::ReadOnly);  // setFileDescriptor internally

    qDebug() << "respPipe opened";

    // FILE* 또는 QFile의 file descriptor 가져오기
    respFd = respPipe->handle();

    // 읽기 가능 상태 감시
    qDebug() << "start watching";
    notifier = new QSocketNotifier(respFd, QSocketNotifier::Read, this);
    connect(notifier, &QSocketNotifier::activated, this, &Widget::handle_pipe_data);
}

void Widget::handle_pipe_data() {
    qDebug() << "handlePipeData called";
    if (!respPipe || !respPipe->isOpen()) return;

    QTextStream in(respPipe);
    QString line = in.readLine();

    if (line.isEmpty()) {
        qDebug() << "읽을 데이터 없음 (EOF?) → 감시 중단";
    }
    else {
        qDebug() << "수신된 데이터:" << line;
    }

    ui->crop_name->setText(line);
    m_client->publish(MQTT_COMMON_TOPIC + "/event/sown", line.toUtf8());

    notifier->setEnabled(false);
    delete notifier;

    respPipe->close();
    delete respPipe;
}
```

```C:\Users\Chanho Kim_PC\Desktop\project\Leafree_AI_Smart_Farm\EM\STT\STTServer.py
import sounddevice as sd
import numpy as np
import librosa
import time
import soundfile as sf
from faster_whisper import WhisperModel
import os
import threading
import io

class STTServer:
    CMD_PIPE = "/tmp/stt_cmd"
    RESP_PIPE = "/tmp/stt_resp"

    def __init__(self, debug=True):
        self.debug = debug
        self.model = WhisperModel("tiny", device="cpu", compute_type="int8")
        self.mic_index = 3
        self.mic_sample_rate = 48000
        self.whisper_sample_rate = 16000
        self.record_duration = 3
        self.play_waiting_sound = False

        if not self.debug:
            self.setup_pipes()

    def setup_pipes(self):
        if not os.path.exists(self.CMD_PIPE):
            os.mkfifo(self.CMD_PIPE)
        if not os.path.exists(self.RESP_PIPE):
            os.mkfifo(self.RESP_PIPE)

    def play_wait_sound(self):
        while self.play_waiting_sound:
            os.system("aplay -D plughw:3,0 wait.wav")

    def play_success_sound(self):
        os.system("aplay -D plughw:3,0 success.wav")

    def play_start_sound(self):
        os.system("aplay -D plughw:3,0 start.wav")
    
    def record_audio(self):
        audio = sd.rec(int(self.record_duration * self.mic_sample_rate),
                       samplerate=self.mic_sample_rate,
                       channels=1,
                       dtype='float32',
                       device=self.mic_index)
        sd.wait()
        return audio.flatten()

    def transcribe_audio(self, audio_data):
        audio_resampled = librosa.resample(audio_data, orig_sr=self.mic_sample_rate, target_sr=self.whisper_sample_rate)
        buffer = io.BytesIO()
        sf.write(buffer, audio_resampled, self.whisper_sample_rate, format='WAV')
        buffer.seek(0)

        print("🚀 [INFO] 추론 시작...")

        self.play_waiting_sound = True
        wait_thread = threading.Thread(target=self.play_wait_sound)
        wait_thread.start()

        start_time = time.time()
        segments, _ = self.model.transcribe(
            buffer,
            language="ko",
            beam_size=1,
            without_timestamps=True
        )
        segments_list = list(segments)
        end_time = time.time()

        self.play_waiting_sound = False
        wait_thread.join()

        print(f"✅ [INFO] 추론 완료! (소요 시간: {end_time - start_time:.2f}초)")

        result_text = ""
        if segments_list:
            self.play_success_sound()
            for segment in segments_list:
                result_text += segment.text + " "
            print(f"📝 Transcription: {result_text.strip()}")
        else:
            result_text = "[인식 실패]"
            print("⚠️ [INFO] 추론은 완료됐지만 인식된 텍스트가 없습니다.")

        return result_text.strip()

    def process_command(self, command):
        if command == "START":
            print("▶️ [INFO] START 명령 수신, STT 실행")
            threading.Thread(target=self.play_start_sound, daemon=True).start()
            audio_data = self.record_audio()
            result = self.transcribe_audio(audio_data)
            return result
        return ""

    def run_pipe_mode(self):
        print("📡 [INFO] Named Pipe 모드 대기 중...")
        while True:
            with open(self.CMD_PIPE, "r") as cmd_pipe:
                command = cmd_pipe.readline().strip()
                result = self.process_command(command)
                
                if result:
                    with open(self.RESP_PIPE, "w") as resp_pipe:
                        resp_pipe.write(result + "\n")

    def run_debug_mode(self):
        print("🛠️ [DEBUG MODE] Enter 키로 STT 실행. Ctrl+C로 종료.")
        while True:
            input("\n▶️ [대기 중] Enter 키를 누르세요...")
            audio_data = self.record_audio()
            self.transcribe_audio(audio_data)

    def start(self):
        try:
            if self.debug:
                self.run_debug_mode()
            else:
                self.run_pipe_mode()
        except KeyboardInterrupt:
            print("\n🛑 [INFO] 프로그램 종료.")

if __name__ == "__main__":
    server = STTServer(debug=False)  # debug=False로 변경 시 Named Pipe 모드
    server.start()
```

```C:\Users\Chanho Kim_PC\Desktop\project\Leafree_AI_Smart_Farm\EM\Water_Solution\Core\Src\freertos.c
void vWaterCTLTask(void *arg)
{
	uint32_t rx_word;
	for (;;)
	{

		if (xQueueReceive(xCtlQ, &rx_word, portMAX_DELAY) == pdPASS)
		{
			WaterCtl_t ctl = { .raw = rx_word };
			Debug_Print("CTL : %d\r\n", ctl.cmd);
			switch (ctl.cmd)
			{
			case 0x01: /* Set Param */
				g_cfg.interval_h    = ctl.interval_h;
				g_cfg.target_adc    = ctl.target_moist;
				g_cfg.last_water_tick = 0;
				break;

			case 0x02: /* Get Param */
				HAL_UART_Transmit(&huart1, (uint8_t *)&g_cfg, sizeof g_cfg, HAL_MAX_DELAY);
				break;

			case 0x03: /* Manual Pump */
				if (ctl.trig != 0U) { xTaskNotifyGive(xWateringTaskHandle); }
				break;

			case 0x04:
				HAL_GPIO_TogglePin(GPIOA, LD2_Pin);

				xSemaphoreTake(xSensorMux, portMAX_DELAY);
				HAL_UART_Transmit(&huart1, (uint8_t *)&g_sensor, sizeof g_sensor, HAL_MAX_DELAY);
				xSemaphoreGive(xSensorMux);
			default:
				break;
			}
		}

	}
}

	uint16_t pair[ADC_BUF_LEN];
	for (;;)
	{
		if (xQueueReceive(xAdcQ, pair, portMAX_DELAY) == pdPASS)
		{
			xSemaphoreTake(xSensorMux, portMAX_DELAY);
			g_sensor.soil_adc    = pair[0] & 0x0FFFU;
			g_sensor.level_adc   = pair[1] & 0x0FFFU;
			g_sensor.sample_tick = xTaskGetTickCount();
			Debug_Print("SOIL : %d, LEVEL : %d\r\n", g_sensor.soil_adc, g_sensor.level_adc);
			xSemaphoreGive(xSensorMux);


			// 토양의 수분 정도가 적다면 워터링 테스크 신호
			if(g_sensor.soil_adc > g_cfg.target_adc)
			{
				xTaskNotify(xWateringTaskHandle, 0, eSetValueWithoutOverwrite);
			}
			// 수위센서의 값이 특정 값보다 작다면 라즈베리파이에 UART로 알림 전송
		}
	}
```
